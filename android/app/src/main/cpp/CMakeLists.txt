cmake_minimum_required(VERSION 3.22)
project(sonora_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(taglib_source URL https://taglib.org/releases/taglib-2.3.2.tar.gz URL_HASH SHA256=3ca2d8afaa7f1cf7f6ed10e511ebc368bfacd6dcaa3dbfa690b89e502e8963dc)
FetchContent_MakeAvailable(taglib_source)
FetchContent_Declare(lame_source URL https://downloads.sourceforge.net/project/lame/lame/4.0/lame-4.0.tar.gz URL_HASH SHA256=3df5124d5ad3a98312ffd7ba6a9b36230e4f8a3e66d3ce0f425e336c32d216eb)
FetchContent_MakeAvailable(lame_source)
set(LAME_FILES VbrTag bitstream encoder fft gain_analysis id3tag lame newmdct presets psymodel quantize quantize_pvt reservoir set_get tables takehiro util vbrquantize version mpglib_interface)
set(LAME_SOURCES "")
foreach(name IN LISTS LAME_FILES)
  list(APPEND LAME_SOURCES "${lame_source_SOURCE_DIR}/libmp3lame/${name}.c")
endforeach()
add_library(mp3lame SHARED ${LAME_SOURCES})
target_include_directories(mp3lame PUBLIC "${lame_source_SOURCE_DIR}/include" PRIVATE "${lame_source_SOURCE_DIR}")
target_compile_definitions(mp3lame PRIVATE STDC_HEADERS=1 HAVE_STDINT_H=1 HAVE_INTTYPES_H=1 HAVE_ERRNO_H=1 HAVE_FCNTL_H=1 HAVE_LIMITS_H=1 HAVE_UNISTD_H=1 HAVE_SYS_TIME_H=1)
target_compile_options(mp3lame PRIVATE "-include" "${CMAKE_CURRENT_SOURCE_DIR}/lame_android_types.h")
target_link_libraries(mp3lame PRIVATE m)
add_library(sonora_audio SHARED sonora_audio.cpp)
target_include_directories(sonora_audio PRIVATE "${taglib_source_SOURCE_DIR}/taglib" "${taglib_source_SOURCE_DIR}/taglib/toolkit" "${taglib_source_SOURCE_DIR}/taglib/mpeg" "${taglib_source_SOURCE_DIR}/taglib/mpeg/id3v2" "${taglib_source_SOURCE_DIR}/taglib/mpeg/id3v2/frames" "${taglib_source_SOURCE_DIR}/taglib/riff" "${taglib_source_SOURCE_DIR}/taglib/riff/wav" "${taglib_source_SOURCE_DIR}/taglib/riff/aiff")
target_link_libraries(sonora_audio PRIVATE tag mp3lame)
target_compile_options(sonora_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(t tag mp3lame sonora_audio)
  target_link_options(${t} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
